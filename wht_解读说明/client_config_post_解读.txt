# 解读版本：当前板端客户端 POST 聚合配置
# 原始文件：C:\Users\DELL\Desktop\4.1 正式文件备份\xquic_file\client_config_post.txt
# 场景判断：POST 上传 + rmnet_data0/usb0 双接口聚合。
# 注意：POST 运行前必须确认板端运行目录下存在 ./download/kkk.txt，否则 3.0 mini_client 会打开文件失败。

# 客户端连接的公网服务端 IP。
server_addr=124.70.219.194

# 服务端 QUIC 端口。
server_port=8443

# request settings
# POST 表示上传请求。
method=POST

# POST 并发流数量。
# 当前配置为 4；代码会根据上传文件大小和 stream_range_size 重新计算实际目标流数量，并限制并发。
stream_cnt=4

# 注释项，不生效。
# 如果启用，表示限制 POST 总发送长度；如果超过实际文件大小，代码会截断到实际文件大小。
#send_data_len=1048576000

# algorithm settings
# 拥塞控制算法：BBR。
cc=bbr

# 多路径调度器：RAP。
mp_sched=rap

# interface list (comma separated, up to MAX_PATH_CNT)
# 双接口聚合上传：rmnet_data0 + usb0。
# 两个接口都需要可用；某一路不通可能导致聚合性能下降或路径创建失败。
interfaces=rmnet_data0,usb0

# kernel socket buffers (bytes)
# 内核发送缓冲区，16 MB。
kernel_sndbuf=16777216

# 内核接收缓冲区，16 MB。
kernel_revbuf=16777216

# user-space buffers (bytes)
# 用户态发送缓冲区，32 KB；POST 上传时较关键。
user_send_buf_size=32768

# 用户态接收缓冲区，32 KB。
user_recv_buf_size=32768

# file settings
# POST 上传源文件路径。
# 该路径相对于 mini_client 的运行目录；文件不存在时客户端会报 failed to open send file。
send_file=./download/kkk.txt

# GET 下载保存路径；POST 场景中保留但通常不作为主体逻辑使用。
download_path=./download/kkk.txt

# 请求目标路径。
# POST 场景下通常作为请求 path；服务端实际落盘位置由服务端 upload_path 控制。
download_target=/upload/ccc.bin

# 关闭 zlog。
use_zlog=0
